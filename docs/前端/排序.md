主要介绍四种排序方法

## 一、冒泡排序

## 代码示例

```js
/**
       * 冒泡排序
       * 复杂度:O(n²)
       * 外层每一次循环都会将当次循环的第n小元素排列好（第一次n=1，第二次n=2...）
       * 内层没必要从0开始，因为没必要自己跟自己比较
       */
function popSort(arr) {
  for (let i = 0; i < arr.length; i++) {
    // 没必要自己跟自己比较啦，所以只需要跟当前元素所在位置开始即可
    for (let j = i; j < arr.length; j++) {
      if (arr[i] > arr[j]) {
        let arr_j = arr[j]; // 先把当前值存起来
        arr[j] = arr[i];
        arr[i] = arr_j;
      }
    }
  }
  return arr;
}
console.log("---------冒泡排序--------");
console.log(popSort([3, 5, 7, 2, 1, 100, 23, 999, -1]));
```

## 二、选择排序

```js
 /**
       * 选择排序
       * 复杂度:O(n²)
       * 第一次选择最小的拍到前面，第二次选择第二小的排到前面,以此类推
       * 是不是有一种四层相识的感觉———没错，就是最后一步不一样
       */
function selectSort(arr) {
  for (let i = 0; i < arr.length; i++) {
    let min_index = i; // 假设最小的下标是当前下标
    for (let j = i; j < arr.length; j++) {
      if (arr[i] > arr[j]) {
        [arr[i], arr[j]] = [arr[j], arr[i]]; // 交换位置嘛
      }
    }
  }
  return arr;
}
console.log("---------选择排序--------");
console.log(selectSort([3, 5, 7, 2, 1, 100, 23, 999, -1]));
```

## 三、归并排序

```js
/**
       * 归并排序
       * 复杂度：O(nlogn)
       * 归并排序是一种分而治之算法。其思想是将原始数组切分成较小的数组，直到每个小数组只有一个位置，接着将小数组归并成较大的数组，直到最后只有一个排序完毕的大数组。
       */
function merge(left, right) {
  var tmp = [];
  while (left.length && right.length) {
    if (left[0] < right[0]) tmp.push(left.shift());
    else tmp.push(right.shift());
  }
  return tmp.concat(left, right);
}

function mergeSort(a) {
  if (a.length === 1) return a;
  var mid = Math.floor(a.length / 2);
  var left = a.slice(0, mid);
  var right = a.slice(mid);
  return merge(mergeSort(left), mergeSort(right));
}
console.log("---------归并排序--------");
console.log(mergeSort([3, 5, 7, 2, 1, 100, 23, 999, -1]));
```

## 四、快速排序

也是复杂度最低的排序

```js
/**
       * 快排,还是递归
       * 它的复杂度为 O(nlog(n))
       * 1. 选择一个基准元素，把数组分为两个子数组。可以随机选择一个元素作为基准元素。
         2. 重排数组以实现小于基准元素的元素被放置在基准的左侧，大于基准元素的元素被放置在右侧。
         3. 对于左侧和右侧的元素重复步骤1和步骤2。也就是递归运行如上步骤，把更大的值和更小的值分割开来。
        */
function quickSort(arr) {
  let baseItem = arr[0]; // 选择一个基准值
  let arr_left = []; // 存放比baseItem小的值
  let arr_right = []; // 存放比baseItem大的值
  if (arr.length < 2) {
    return arr;
  } else {
    for (let i = 0; i < arr.length; i++) {
      if (arr[i] > baseItem) {
        arr_right.push(arr[i]);
      } else if (arr[i] < baseItem) {
        arr_left.push(arr[i]);
      }
    }
  }
  return quickSort(arr_left).concat(baseItem, quickSort(arr_right));
}
console.log("---------快速排序--------");
console.log(quickSort([3, 5, 7, 2, 1, 100, 23, 999, -1]));
```
