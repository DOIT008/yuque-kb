## 普通队列

## 特点

FIFO（先入先出），可以想象是排队等车，排在前面的肯定第一个上车呀！

## 模拟实现

### 基于数组

## 代码示例

```js
//  基于数组实现
class Queue {
  constructor() {
    this.queue = [];
  }
  // 是否为空
  isEmpty() {
    return !this.queue.length;
  }
  //入列：像队列尾部添加元素
  enterQueue(el) {
    this.queue.push(el);
  }
  // 出列：按照“先入后出”的原则，每次取出第一个元素
  dequeue() {
    return this.queue.shift();
  }
  // 返回队列的第一个元素
  front() {
    return this.queue[0];
  }
  // 队列的大小
  size() {
    return this.queue.length;
  }
  // 将栈以字符串的形式返回
  toString() {
    return this.queue.join(" ");
  }
}
let queue  = new Queue();
queue.enterQueue(1)
queue.enterQueue(19)
queue.enterQueue(89)
queue.enterQueue(10)
queue.enterQueue(18)
queue.dequeue()
console.log(queue.isEmpty()); //false
console.log(queue.size());// 4
console.log(queue.front()); // 19
console.log(queue.toString()); // 19 89 10 18
```

### 基于对象

```js
    class Queue {
      constructor(){
        this.count = 0; // 当前队列中元素的个数
        this.lowestCount = 0;// 当前排在最前面的元素的下标，默认是0
        this.items = {}; // 元素的集合 
      }
      // 添加元素
      enqueue(element){
        this.items[this.count+this.lowestCount] = element;// 因为第一个下标不一定是0
        this.count++
      }
      // 删除最前面的元素，并返回之,并不一定是最先添加进来的元素，
      dequeue(){
        if(this.isEmpty()){
          return undefined
        }else{
          let element = this.items[this.lowestCount];// 保存要被删除的元素
          delete this.items[this.lowestCount]; // 删除元素
          this.lowestCount++;// 删除一个元素后，最低的那个元素下标肯定要增加的
          this.count--;// 删除一个元素，元素的个数肯定要减一的呀
          return element
        }
      }
      // 返回队列中第一个元素，只是返回
      peek(){
        if(this.isEmpty()){
          return undefined
        }else{
          return this.items[this.lowestCount]
        }
      }
      // 判断是否为空
      isEmpty(){
        return this.count===0
      }
      // 返回队列中包含的元素个数
      size(){
        return this.count
      }
      // 清空队列
      clear(){
        this.items = {};
        this.count = 0;
        this.lowestCount = 0
      }
      // 字符串化
      toString(){
        if(this.isEmpty()){
          return ''
        }else{
          let str = this.items[this.lowestCount]
          for (let i = this.lowestCount+1; i < this.lowestCount+this.count; i++) {
            str+=`${this.items[i]}`
          }
          return str
        }
      }
    }
    const queue = new Queue();
    let str = '博学之，审问之，慎思之，明辨之，笃行之';
    for (let i = 0; i < str.length; i++) {
      queue.enqueue(str[i])
    }
    console.log(queue);
    queue.dequeue();
    queue.dequeue();
    queue.dequeue();
    queue.dequeue();
    console.log(queue);
    queue.enqueue('啦啦啦')
    queue.enqueue('哈哈哈')
    console.log(queue);
    console.log(queue.peek());
    queue.dequeue();
    console.log(queue.peek());
    console.log(queue.isEmpty());
    console.log(queue.size());
    console.log(queue.toString());
```

## 双端队列

## 特点

双端队列和普通队列最大的不同在于，它允许我们在队列的头尾两端都能在O(1)的时间内进行数据的查看、添加和删除。

## 实现

与队列相似，我们可以利用一个双链表实现双端队列。

```js
    class DeQue{
      constructor(){
        this.count = 0;
        this.lowestCount = 0;
        this.items = {}
      }
      // 在队列前端添加元素
      addFront(element){
        if(this.isEmpty()){
          this.addBack(element)
        }else if(this.lowestCount>0){
          this.lowestCount--;
          this.items[this.lowestCount]= element
        }else{
          // this.items[0] = element;
          for (let i = this.count; i >0; i--) {
            this.items[i] = this.items[i-1]
          }
          this.items[0] = element;// 设置最前面的元素
          this.count++;
          this.lowestCount = 0
        }
      }
      // 在队列的后面添加元素
      addBack(element){
        this.items[this.count] = element;
        this.count++
      }
      // 在前面移除一个元素,并返回之
      removeFront(){
        if(this.isEmpty()){
          return undefined
        }else{
          let frontElement = this.items[this.lowestCount];
          delete this.items[this.lowestCount];
          this.lowestCount++
          return frontElement
        }
      }
      // 在后面移除一个元素
      removeBack(){
        if(this.isEmpty()){
          return undefined
        }else{
          let backElement = this.items[this.count-1];
          delete this.items[this.count];
          this.count--
          return backElement
        }
      }
       // 返回双端队列中的前端第一个元素，只是返回
      peekFront(){
        if(this.isEmpty()){
          return undefined
        }else{
          return this.items[this.lowestCount]
        }
      }
      // 返回双端队列中后端第一个元素
      peekBack(){
        if(this.isEmpty()){
          return undefined
        }else{
          return this.items[this.count-1]
        }
      }
      // 判断是否为空
      isEmpty(){
        return this.count- this.lowestCount===0
      }
      // 返回队列中包含的元素个数
      size(){
        return this.count-this.lowestCount
      }
      // 清空队列
      clear(){
        this.items = {};
        this.count = 0;
        this.lowestCount = 0
      }
    }
```

## 应用场景

双端队列最常用的地方就是实现一个长度动态变化的窗口或者连续区间，而动态窗口这种数据结构在很多题目中都有运用。

**典型应用题目**：击鼓传花（循环队列），回文字符串。

## 优先级队列

## 特点

一般的队列在插入数据的时候直接插入到队列的后面就行了，但是对于优先级队列，在插入数据的时候要先考虑**这个数据的优先级，在和已有数据的优先级进行比较过后，才能确定该数据的位置，**队列的其他处理方式和普通队列一样。

## 实现

### 数组

```js
 /**
 * @description: 封装元素类
 * @return {*} Object
 */
class QueueElement {
  constructor(element, priority) {
    this.element = element;
    this.priority = priority;
  }
}
//  基于数组实现
class PriorityQueue {
  constructor() {
    this.priorityQueue = [];
  }
  // 是否为空
  isEmpty() {
    return !this.priorityQueue.length;
  }
  //入列：像队列尾部添加元素
  enterQueue(el, priority) {
    let element = new QueueElement(el, priority);
    // 如果队列为空，直接push
    if (this.priorityQueue.length === 0) {
      this.priorityQueue.push(element);
    } else {
      // 默认下标
      let index = -1;
      // 遍历比较
      for (let i = 0; i < this.priorityQueue.length; i++) {
        const comparedEle = this.priorityQueue[i];
        // 找到第一个比要插入的元素优先级小的元素的下标，并在此处插入
        if (priority < comparedEle.priority) {
          index = i
          this.priorityQueue.splice(index, 0, element);
          break;
        }
      }
      // 说明在队列中没找到比要插入元素的权重小的，直接插入到最后
      if(index===-1){
        this.priorityQueue.push(element)
      }
    }
  }
  // 出列：按照“先入后出”的原则，每次取出第一个元素
  dequeue() {
    return this.priorityQueue.shift();
  }
  // 返回队列的第一个元素
  front() {
    return this.priorityQueue[0];
  }
  // 队列的大小
  size() {
    return this.priorityQueue.length;
  }
  // 将栈以字符串的形式返回
  toString() {
    return this.priorityQueue.join(" ");
  }
}
let priorityQueue = new PriorityQueue();
priorityQueue.enterQueue("丫丫", 1);
priorityQueue.enterQueue("晶晶", 0.3);
priorityQueue.enterQueue("贝贝", 0.5);
priorityQueue.enterQueue("欢欢", 1.2);
priorityQueue.enterQueue("迎迎", 0.7);
priorityQueue.enterQueue("妮妮", 1.1);
console.log(priorityQueue.toString());
```
