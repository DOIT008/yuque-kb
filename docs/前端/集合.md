## 一、定义

集合是由一组**无序且唯一（即不能重复）**的项组成的， 比如说，一个由大于或等于 0 的整数组成的自然数集合：N = {0, 1, 2, 3, 4, 5, 6, …}。集合中的对象列表用花括号（`{}`）包围。

还有一个概念叫空集。空集就是不包含任何元素的集合 ，也就是`{}`。

## 二、实现

我们使用`class`类来实现一个`Set`。

## 代码示例

```js
    class Set {
      constructor(){
        this.items = {}
      }
      // 添加元素
      add(element){
        if(!this.has(element)){
          this.items[element] = element;
          return true
        }
        return false
      }
      // 删除元素
      delete(element){
        if(this.has(element)){
          delete this.items[element]
        }
      }
      // 判断某个元素是否存在
      has(element){
        return element in this.items
      }
      // 清空集合所有元素
      clear(){
        this.items = {}
      }
      // 返回集合所包含的元素数量
      size(){
        return Object.keys(this.items).length;
      }
      // 返回由集合元素组成的数组
      values(){
        let arr = [];
        for (const key in this.items) {
          arr.push(this.items[key])
        }
        return arr
        // return Object.values(this.items)
      }
    }
    let set = new Set();
    set.add(12)
    set.add('哈哈')
    set.add(true)
    set.add(1)
    set.add(90);
    console.log(set);
    console.log(set.values());
    console.log(set.size());
    console.log(set.has(12));
    console.log(set.has(11));
    console.log(set.has('哈哈'));
    set.delete(12);
    set.delete(14);
    console.log(set);
```

## 三、计算

基于上述的实现添加集合的常见运算：**交集，并集，差集，是否是子集。**

```js
{
          // 并集；A U B
        union(otherSet) {
          let arr = [...otherSet.values(), ...this.values()];
          let newItems = new Set();
          for (let i = 0; i < arr.length; i++) {
            if (!newItems.has(arr[i])) {
              newItems.add(arr[i]);
            }
          }
          return newItems;
        }
        // 交集； A ∩ B
        intersection(otherSet) {
          let arr = [...otherSet.values(), ...this.values()];
          let newItems = new Set();
          for (let i = 0; i < arr.length; i++) {
            // 同时存在于A，B
            if (otherSet.has(arr[i]) && this.has(arr[i])) {
              // 并且newItems中还没有
              if (!newItems.has(arr[i])) {
                newItems.add(arr[i]);
              }
            }
          }
          return newItems
        }
        // 差集，A中存在，B中不存在
        difference(){
          let arr = [...otherSet.values(), ...this.values()];
          let newItems = new Set();
          for (let i = 0; i < arr.length; i++) {
            // 同时存在于A，B
            if (!otherSet.has(arr[i]) && this.has(arr[i])) {
              // 并且newItems中还没有
              if (!newItems.has(arr[i])) {
                newItems.add(arr[i]);
              }
            }
          }
          return newItems
        }
        // A是otherSet的子集
        isSubsetOf(otherSet){
          // 空集是任何集合的子集（包括空集）
          if(!this.size()&&!otherSet.size()){
            return true
          }
          let isSubset = true;
          let arr = this.values();// 拿到所有的元素集合进行遍历
          if(arr.length){
            isSubset = arr.every(item=>otherSet.has(item));
          }
          return isSubset
        }
}
```
