通常我们写组件的时候，一般写的都是模板组件，但是有时候可能写函数式组件会更有效率，什么是函数式组件呢，其实就是在写组件的时候使用`render`函数，具体的可以像这样子：

## 代码示例

```js
// main.js
import { createApp, reactive, toRefs,h } from "vue"; // 引入createApp函数
import App from "./App.vue"; // 其实其实就是一个对象
import "./index.css";
// 执行挂载进行实例化

const app = createApp(App);
// 函数式组件
app.component("my-component", {
  // props:object,array,
  //emits: ['update:modelValue'],监听的事件
  render() {
    return h(...);
  },
});

app.mount("#app");

```

这里仅仅对`h(...)`函数进行一下说明，里面式一些参数，具体的参数如下：

```vue
// @returns {VNode}
h(
  // {String | Object | Function} tag
  // 一个 HTML 标签名、一个组件(引入的对象)、一个异步组件、或
  // 一个函数式组件。
  //
  // 必需的。
  'div',

  // {Object} props
  // 与 attribute、prop 和事件相对应的对象。
  // 我们会在模板中使用。
  //
  // 可选的。如果没有，为了避免歧义，可以传null
  {},

  // {String | Array | Object} children
  // 子 VNodes, 使用 `h()` 构建,
  // 或使用字符串获取 "文本 Vnode" 或者
  // 有插槽的对象。
  //
  // 可选的。
  [
    'Some text comes first.',
    h('h1', 'A headline'),
    h(MyComponent, {
      someProp: 'foobar'
    })
  ]
)
```

第二个参数进行一次着重说明，里面会包含一些选项,**所有的选项都是展平的（没有嵌套的属性，包括事件啦，属性了，class，style等都一样）**

```json
// Vue3.x 变化的部分
{
	// v-model
	modelValue: this.modelValue,
  // 监听的自定义事件
  'onUpdate:modelValue': value => this.$emit('update:modelValue', value)
  // 正常绑定的事件 on事件名
  onClick: $event => console.log('clicked', $event.target)
	// 带修饰符的事件名,.capture,.once,.once.capture
	onClickCapture: this.doThisInCapturingMode,
  onKeyupOnce: this.doThisOnce,
  onMouseoverOnceCapture: this.doThisOnceInCapturingMode
}

// 其他属性或选项，基本上直接写就行了，而不是像Vue2.x中那样要写在对应的选项里
{

}
```

其实第三个参数也可以插槽

```js
render() {
  // `<div><slot></slot></div>`
  return h('div', {}, this.$slots.default())
}
```

## 举个例子

```js
app.component("my-component", {
  render() {
    return h("div", null, [
      h("h1", null, [
        h(
          "a",
          {
            href: "www.baidu.com",
            // class: "zhangsan lisi",// 可以是字符串

              zahnsgan: true,

            id: "baiDu",
            name: "百度一下，你就知道",
          },
          "百度一下"
        ),
      ]),
    ]);
  },
});
```

## 再举个例子

### 基本例子

```vue
// TestFunc.vue
<script>
// 引入h函数
import { h } from "vue";
const TestFunc = function (props, ctx) {
  console.log(props, ctx);
  return h("div", {
    class: "container",
  },"先测试一下再说");
};
export default TestFunc;
</script>
<div></div>
```

### 全面的例子

```vue
// TestFunc.vue
<script>
import { h, defineEmit } from "vue";
const TestFunc = function (props, ctx) {
  /**
   * props:相当于以前的props对象,访问某个传递过来的数据X需要通过props.xx的形式访问
   * ctx：也是一个对象，包含attrs，slots和emit方法
   * 注意：函数式组件没有生命周期
   *  */
  console.log(props, ctx);
  const emits = defineEmit(["testEmit"]); // 定义压迫抛出的事件
  return h(
    "div",
    {
      class: "container",
    },
    [
      h("h" + props.index, {}, "哈哈啥我是测试的"),
      h(
        "div",
        {
          class: "innerContent",
          style: {
            color: "red",
          },
        },
        [
          "我来自内部",
          h(
            "button",
            {
              class: "test-btn",
              id: "testBtn",
              onClick() {
                console.log(props.index); //2
                console.log(ctx.slots.default()[0].children.trim("")); // 拿到默认插槽的内容
                console.log(ctx.slots.nameSlot()[0].children); //拿到具名插槽nameSlot的内容
                console.log(ctx.slots.nameSlotIndex()[0].children); //拿到具名插槽nameSlotIndex的内容
                ctx.emit("testEmit", "嘿嘿"); //嘿嘿
              },
            },
            "测试按钮"
          ),
          h(
            "a",
            {
              name: "link",
              href: "http://www.baidu.com",
            },
            "百度一下，你就知道"
          ),
        ]
      ),
    ]
  );
};
export default TestFunc;
</script>
```
