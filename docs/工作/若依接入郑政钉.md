## 二、代码层面

在若依的框架下实现郑政钉登录解决方案：

```js
router.beforeEach((to,from,next)=>{
   if (to.path === '/login') {
     // 郑政钉登录会在url后面拼上auth_code
      if(to.query.auth_code){
        next()
      }else{
        next({ path: '/' })
      }
      NProgress.done()
    } 
})
```

 

```js
created() {
  // 如果登录界面有auth_code，就走政政钉登录，内部的登录逻辑和正常登录逻辑一样
    if (this.$route.query.auth_code) {
      this.$store.dispatch("LoginByAuthCode", this.$route.query.auth_code).then(() => {
        this.$router.push({ path: this.redirect || "/" }).catch(() => { })
      }).catch(() => {
        this.loading = false
      })
    }else{
      this.getCode()
      this.getCookie()
    }
  },
```

 

```js
// 登录
  Login({ commit }, userInfo) {
    const username = userInfo.username.trim()
    const password = userInfo.password
    const code = userInfo.code
    const uuid = userInfo.uuid
    return new Promise((resolve, reject) => {
      login(username, password, code, uuid).then(res => {
        setToken(res.token)
        commit('SET_TOKEN', res.token)
        resolve()
      }).catch(error => {
        reject(error)
      })
    })
  },
  // 郑政钉登录
  LoginByAuthCode({ commit }, authCode) {
    return new Promise((resolve, reject) => {
      console.log('LoginByAuthCode', authCode)
      loginByAuthCode(authCode).then(res => {
        setToken(res.token)
        commit('SET_TOKEN', res.token)
        resolve()
      }).catch(error => {
        reject(error)
      })
    })
  },
```

 

```js
// 钉钉登录
export function loginByAuthCode(authCode) {
  return request({
    url: '/dingTalkLogin',
    headers: {
      isToken: false
    },
    method: 'post',
    data: {
      authCode: authCode
    }
  })
}
```
