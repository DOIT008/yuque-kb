## 代码示例

```vue
<el-upload
  class="avatar-uploader"
  :action="url" // 上传路径
  name="file" // 文件名
  :headers="uploadHeader"
  :show-file-list="true"
  list-type="picture-card"
  :on-remove="handleRemove"
  :file-list="authForm.credentialUrls"
  :on-success="handleAvatarSuccess"
  :before-upload="beforeAvatarUpload"
>
  <i class="el-icon-plus avatar-uploader-icon"></i>
</el-upload>

 handleAvatarSuccess(res, file, fileList) {
      this.authForm.credentialUrls.push({
        name:file.name,
        url: file.response.data.fullFileName //图片全路径url
      })
    },
    // 移除已上传的文件
    handleRemove(file, fileList) {
      this.authForm.credentialUrls = fileList; 
    },

 //上传前钩子
   beforeAvatarUpload(file) {
      const isLt20M = file.size / 1024 / 1024 < 20;

      if (!isLt20M) {
        this.$message.error("上传头像图片大小不能超过 20MB!");
      }
      return isLt20M;
    },

      //获取认证详情---用于回显
    getIdentityInfo() {
      this.$http
        .get("/judicial/authDetail")
        .then(({ data: res }) => {
          // 只有这样才会正常回显
          this.authForm.credentialUrls = res.data.credentialUrls.map(item=>{
            return {
              url: item, // 图片全路径url
              name:item
            }
          })
          if (res.code !== 0) {
            return this.$message.error(res.msg);
          }
        })
        .catch(() => {});
    },
```

## 三、导出word、excel

```js
axios({
        method: 'POST',
        url: `${window.SITE_CONFIG['apiURL']}/api/service/export?apiId=${row.apiId}&applyCode=${row.applyCode}`,
        responseType: 'blob', // 很重要
        headers: {
          token: Cookies.get('token'),
          'Content-Type': 'application/json; charset=UTF-8',
        },
      }).then((res) => {
        // 导出word 
        const blob = new Blob([res.data], {
          type: 'application/msword;charset=UTF-8',
        });
        // 导出excel
        // const blob = new Blob([res.data], {
        //   type: 'application/vnd.ms-excel;charset=UTF-8'
        // });
        const url = window.URL.createObjectURL(blob);
        let link = document.createElement('a');
        link.style.display = 'none'; 
        link.href = url;
        link.setAttribute('download', `${row.serviceName}.doc`);
        // link.setAttribute('download', `${row.serviceName}.xls`);
        document.body.appendChild(link);
        link.click();
        URL.revokeObjectURL(link.href); // 释放URL 对象
        document.body.removeChild(link);
        link = null;
      });
```

## 五、Vant 上拉加载下拉刷新

```vue
<template>
  <div class="invoicing-order">
    <van-nav-bar title="待开票订单" left-text="返回" left-arrow @click-left="$router.go(-1)" />
    <van-pull-refresh v-model="refreshing" @refresh="onRefresh">
      <van-list class="order-list" v-model="loading" :finished="finished" finished-text="没有更多了" @load="onLoad"
        :immediate-check="false" :offset="20">
        <div class="order-item" v-for="(item,index) in orderList" :key="index" :title="item">
          <div class="plate-number">
            <span>车牌号：</span><span class="plate-number-text">{{ item.carNum }}</span>
            <span class="status"
              :class="{ 'waited': item.operateStatus =<mark> '0', 'opening': item.operateStatus </mark>= '1', 'success': item.operateStatus =<mark> '2', 'failed': item.operateStatus </mark>= '3' }">{{
  item.operateStatus | operateStatusFilter }}</span>
          </div>
          <div class="address">
            <span>地址：</span><span class="address-text">苏州市吴中区吴中大道100号</span>
          </div>
          <div class="specify">
            <span>自驾审车</span>
          </div>
          <div class="money">
            <span>实付金额：</span><span class="money-text">￥100.00</span>
          </div>
          <van-checkbox class="check-box" v-model="item.checked"></van-checkbox>
        </div>
      </van-list>
    </van-pull-refresh>
    <van-button type="primary" round class="btn" @click="navigateInvoicingOrderForm">确定</van-button>
  </div>
</template>

<script>
import { getInvoiceOrderList } from '@/api/order'
export default {
  name: 'InvoicingOrder',
  data () {
    return {
      orderList: [],
      count: 0,
      loading: false,
      refreshing: false,
      finished: false,
      pageNum: 1,
      pageSize: 10
    }
  },
  filters: {
    operateStatusFilter (value) {
      const operateStatusMap = {
        0: '待开票',
        1: '开票中',
        2: '已开票',
        3: '开票失败'
      }
      return operateStatusMap[value]
    }
  },
  mounted () {
    this.getOrderList()
  },
  methods: {
    navigateInvoicingOrderForm () {
      this.$router.push({
        path: '/order/invoicing-order-form'
      })
    },
    /**
     * 获取订单列表
     *
     * @returns 无返回值
     */
    getOrderList () {
      const params = {
        operateStatus: 0,
        pageNum: this.pageNum,
        pageSize: this.pageSize
      }
      getInvoiceOrderList(params).then(res => {
        if (res.code === 200) {
          if (this.pageNum === 1) {
            this.orderList = res.rows
          } else {
            this.orderList = this.orderList.concat(res.rows)
          }
          if (res.rows.length < this.pageSize) {
            this.finished = true
          }
          // 数据完成加载后，重置 loading 和 refreshing
          this.loading = false
          this.refreshing = false
        }
      }).catch(err => {
        console.log(err)
      })
    },
    /**
     * 下拉刷新
     *
     * @returns 无返回值
     */
    onRefresh () {
      console.log('refresh')
      this.refreshing = true
      this.loading = true
      this.finished = false
      this.pageNum = 1
      this.getOrderList()
    },
    /**
     * 上拉加载
     *
     * @returns 无返回值
     */
    onLoad () {
      console.log('load')
      this.loading = true
      this.pageNum++
      this.getOrderList()
    }
  }
}
</script>
```
