## 定义

链表就像火车一样有，火车头（head）和车厢（node），每个节点都由元素本身（item）和指向下一个节点的指针组成，就像下面这这样——

![image.png](/images/695e36083cedcab3.png)

## 优缺点

## 优点

1.  **插入/删除效率高**：根据链表的特点，你要插入一个节点，你改变的只是这个节点前后的节点（准确说是指向），所以会非常快，时间复杂度是O(1)。
2.  内存空间不必是连续的，可以充分利用计算机的内存实现灵活**动态内存管理**。
3.  链表在创建的时候**不必确定空间大小**，可以无限的延伸下去。

## 缺点

1.  **读取数据慢**：你必须**从头开始一个节点一个节点**地查才能找到目标节点，如果链表比较长性能就比较低下了。

## 代码示例

```js
function LinkedList(params) {
  // 头部,默认是null
  this.head = null;
  // 疯转节点
  function Node(data) {
    this.data = data; //数据本身
    this.next = null; // 刚开始不知道指向哪个节点，所以是null
  }
```

```js
      class LinkedList {
        constructor() {
          this.head = undefined; // 初始化的时候head指向肯定是空呀
          this.count = 0;
        }
        push(element) {
          let node = new Node(element); // 创建一个节点
          if (this.head === undefined) {
            // 如果是空链表就直接将该节点给head
            this.head = node;
          } else {
            // 如果不是非空链表，就需要一个个的查找，因为不知道要查找多少次，只能使用while
            let current = this.head;
            while (current.next !== null) {
              current = current.next; //寻找下一个节点
            }
            current.next = node;
          }
          this.count++; // 添加新节点，count肯定要++的
        }
        // 删除元素
        removeAt(index) {
          // 说明移除的是第一个元素
          if (index >= 0 && index <= this.count) {
            // 如果删除的是第一个
            let currentNode = this.head;
            if (index === 0) {
              let node = currentNode.next; // 第二个元素
              this.head = node;
            } else {
              let preNode = this.getElement(index - 1); // 找到当前元素的上一个元素
              currentNode = preNode.next; // 当前位置的元素，也就是要删除的元素
              preNode.next = currentNode.next; // 前一个元素的next指向新的元素
            }
            this.count--;
            return currentNode.element;
          } else {
            return undefined;
          }
        }
        // 找到元素
        getElement(index) {
          if (index >= 0 && index <= this.count) {
            if (index === 0) {
              return this.head.element;
            } else {
              let currentNode = this.head;
              for (let i = 0; i < index; i++) {
                currentNode = currentNode.next; // 这就是当前的节点
              }
              return currentNode;
            }
          } else {
            return undefined;
          }
        }
        // 在任意节点插入元素
        insert(element, index) {
          // 越界
          if (index >= 0 && index <= this.count) {
            let node = new Node(element); // 创建节点实例
            let currentNode = this.head;
            // 如果插入的位置是第一个元素
            if (index === 0) {
              this.head = node;
              node.next = currentNode.next;
            } else {
              let preNode = this.getElement(index - 1); // 找到当前元素的上一个元素
              let currentNode = preNode.next; // 当前位置的元素
              preNode.next = node; // 前一个元素的next指向新的元素
              node.next = currentNode; // 新元素的next指向原来位置的元素
            }
            this.count++;
          }
        }
        // 查找节点所在的位置，如果找不到返回-1
        indexOf(element) {
          let currentNode = this.head;
          let val = currentNode.element;
          // 如果匹配到第一个元素
          if (val === element) {
            return 0;
          } else {
            // 否则从第二个元素开始循环
            let index = 1;
            while (index < this.count) {
              currentNode = currentNode.next;
              val = currentNode.element;
              if (val === element) {
                return index;
              }
              index++;
            }
            return -1;
          }
        }
        // 删除某个元素
        remove(element) {
          const index = this.indexOf(element); // 找到当前元素
          this.removeAt(index); // 删除操作，利用已有的api
        }
        // 链表是否为空，只需要查看内部元素个数即可
        isEmpty() {
          return this.count === 0;
        }
        // 查看大小
        size() {
          return this.count;
        }
        //
        getHead() {
          return this.head;
        }
        // 返回element拼接的字符串
        toString() {
          let str = "";
          let currentNode = this.head;
          if (this.count == 0) {
            return str;
          } else {
            for (let i = 0; i < this.count; i++) {
              str += `-->${currentNode.element}`; //当前节点
              currentNode = currentNode.next;
            }
          }
          return str;
        }
      }

        constructor(element) {
          this.element = element;
          this.next = null;
        }
      }
```

## 2\. 双向链表

在单向链表的基础上，实现一些主要方法；

```js
   // 双向链表
      /**
       * 实现双向链表节点
       */
      class DoubleNode extends Node {
        constructor(element) {
          super(element);
          this.prev = null;
        }
      }

      /**
       * 实现链表本身
       */
      class DoubleLinkedList extends LinkedList {
        constructor() {
          super();
          this.tail = null; // 永远指向最后一个元素
        }
        // 寻找某一个节点，如果有就返回，没有返回undefined；
        getElement(index) {
          if (index >= 0 && index <= this.count) {
            let targetNode = this.head;
            for (let i = 0; i < index; i++) {
              targetNode = targetNode.next;
            }
            return targetNode;
          } else {
            return undefined;
          }
        }
        // push 往链表中添加节点
        push(element) {
          let doubleNode = new DoubleNode(element); // 创建一个节点
          if (this.count === 0) {
            this.head = doubleNode;
          } else {
            // 找到最后一个元素
            let node_index = this.getElement(this.count - 1); //拿到的只是一个指针
            node_index.next = doubleNode; // next指向当前的元素
            doubleNode.prev = node_index; // 当前元素的pre 指向前一个元素
          }
          this.tail = doubleNode;
          this.count++;
        }
        /**
         * insert，在任意位置插入新的元素：
         * 找到目前该位置的元素和前一个元素，修改各自的pre和next指向即可
         * 三种特殊位置，index = 0，index=count
         * */
        insert(element, index) {
          if (index >= 0 && index <= this.count) {
            let insertNode = new DoubleNode(element);
            if (index === 0) {
              if (this.head) {
                let headNextNode = this.head; // 找到此时此刻index=0位置的元素
                this.head = insertNode; // head指向新元素
                insertNode.next = headNextNode; //
                headNextNode.prev = insertNode; // 原位置元素的prev指向新创建的元素;
              } else {
                this.tail = insertNode;
                this.head = insertNode;
              }
            } else if (index === this.count) {
              // 最后一项
              let currentNode = this.getElement(index - 1); // 找到前一个元素
              this.tail = insertNode;
              currentNode.next = insertNode; // 当前元素的next指向新元素
              insertNode.prev = currentNode; // 新元素的prev指向前一个元素
            } else {
              let preNode = this.getElement(index - 1); // 找到前一个元素
              let currentNode = this.getElement(index); // 找到当前元素
              preNode.next = insertNode; // 前一个元素的next指向新元素
              insertNode.prev = preNode; // 新元素的pre指向前一个元素
              insertNode.next = currentNode; // 新元素的next指向原位置元素
              currentNode.prev = insertNode; // 原位置元素的prev指向新元素
            }
            this.count++;
            return true;
          } else {
            return false;
          }
        }
        // 从任意位置移除元素,并返回
        removeAt(index) {
          if (index >= 0 && index <= this.count) {
            if (index === 0) {
              this.count--
              let headNode = this.head;
              if (this.head) {
                if (this.count >= 2) {
                  let headNextNode = this.head.next; // 找到第二个元素
                  this.head = headNextNode; // head 指向第二个元素
                  headNextNode.prev = null; //第二个元素的prev指向head，所以是null
                  return headNode;
                } else {
                  this.head = null;
                  return headNode;
                }
              } else {
                return undefined;
              }
            } else if (index === this.count) {
              let currentNode = this.getElement(index-1);// 找到要删除的元素,也就是最后一个元素
              let prevNode = currentNode.prev;// 前一个元素
              this.tail = prevNode;
              prevNode.next = null;
              this.count--;
              return currentNode
            }else{
              let currentNode = this.getElement(index);// 找到要删除的当前元素;
              let prevNode = currentNode.prev;// 要删除元素的前一个元素
              let nextNode = currentNode.next;// 要删除元素的后一个元素
              prevNode.next = currentNode.next;// 将前一个元素的next指向要删除元素的下一个元素
              nextNode.prev = prevNode;// 将后一个元素的prev指向前一个元素
              this.count--;
              return currentNode
            }
          }
        }
      }
      let doubleLinkedList = new DoubleLinkedList();
      doubleLinkedList.push("周一");
      doubleLinkedList.push("周二");
      doubleLinkedList.push("周三");
      doubleLinkedList.insert("周日", 0);
      doubleLinkedList.insert("周八", 4);
      doubleLinkedList.insert("周五", 2);
      // 周日-周一-周五-周二-周三-周八
      console.log(doubleLinkedList.removeAt(6));
      console.log(doubleLinkedList.removeAt(0));
      console.log(doubleLinkedList.removeAt(2));
```

## 3\. 循环链表

循环链表可以像链表一样只有**单向引用**，也可以像双向链表一样有双向引用。循环链表和链表之间唯一的区别在于，**最后一个元素的next指向第一个元素（head）**

## 4.有序链表

有序链表是指保持元素有序的链表结构

## 使用链表实现栈

```js
class StackLinkedList { 
 constructor() { 
 this.items = new DoubleLinkedList(); // {1} 
 } 
 push(element) { 
 this.items.push(element); // {2} 
 } 
 pop() { 
 if (this.isEmpty()) { 
 return undefined; 
 } 
 return this.items.removeAt(this.size() - 1); // {3} 
 } 
}
```

**为啥要用双向链表而不是单链表？** `DoubleLinkedList` 类有列表最 后一个元素（`tail` ）的引用，无须迭代整个链表的元素就能获取它。双向链表可以直接获取头 （说实话有点牵强）
