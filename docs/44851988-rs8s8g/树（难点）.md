一个树结构包含一系列存在父子关系的节点。每个节点都有一个父节点（除了顶部的第一个 节点）以及零个或多个子节点 ，如下图所示：

![image.png](/images/8b47c51291a77c61.png)

节点的一个属性是**深度**，节点的深度取决于它的祖先节点的数量。比如，节点 3 有 3 个祖先 节点（5、7 和 11），它的深度为 3。 树的**高度**取决于所有节点深度的最大值。

## 一、二叉树

二叉树中的节点最多只能有两个子节点：一个是左侧子节点，另一个是右侧子节点。 **二叉搜索树（BST）**是二叉树的一种，但是只允许你在左侧节点存储（比父节点）小的值， 在右侧节点存储（比父节点）大的值 。

和链表一样，我们将通过**指针**（引用）来表示节点之间的关系，在树中所谓节点又被称为**键。**

## 代码示例

```js
      // 创建一个树节点类
      class Node {
        constructor(key) {
          this.key = key;
          this.left = null;
          this.right = null;
        }
      }
      // 比较节点的值是否相同
      function compare(key, base) {
        if (key > base) {
          return "MORE_THAN";
        } else {
          return "LESS_THAN";
        }
      }

      class BinarySearchTree {
        constructor() {
          this.compareFn = compare;
          this.root = null; // 初始化根节点是null
        }
        // 插入节点
        insert(key) {
          if (this.root == null) {
            // 如果没有根节点那么首先插入到根节点的位置
            this.root = new Node(key);
          } else {
            // 如果有根节点则执行插入操作
            this.insertNode(this.root, key);
          }
        }
        // 这个实现要根据二叉树的特点
        insertNode(node, key) {
          if (this.compareFn(key, node.key) === "LESS_THAN") {
            // 如果要插入的节点比当前节点小
            if (node.left === null) {
              // 如果左侧是null，则直接插入到这个位置
              node.left = new Node(key);
            } else {
              this.insertNode(node.left, key); // 如果左侧有节点，则将node.left传入来继续寻找
            }
          } else {
            if (node.right === null) {
              // 如果右侧是null，就插入到右侧节点位置
              node.right = new Node(key);
            } else {
              this.insertNode(node.right, key);// 如果右侧有节点，则将node.right传入来继续寻找
            }
          }
        }
        // 1. 中序遍历
        inOrderTraverse(callBack){
          this.inOrderTraverseNode(this.root,callBack)
        }
        // 真正的遍历操作
        inOrderTraverseNode(node,callBack){
          // 如果当前节点存在
          if(node!==null){
            this.inOrderTraverseNode(node.left,callBack);
            callBack(node.key)
            this.inOrderTraverseNode(node.right,callBack)
          }
        }
      }
      let binarySearchTree = new BinarySearchTree();
      binarySearchTree.insert(19);
      binarySearchTree.insert(8);
      binarySearchTree.insert(22);
      binarySearchTree.insert(32);
      binarySearchTree.insert(11);
     

      function print(val){
        console.log(val);
      }
      binarySearchTree.inOrderTraverse(print)
```

## 1.3、遍历排序

访问树的所有节点有三种方 式：**中序、先序和后序。**

**上面实现的**`inOrderTraverseNode`就是中序遍历，为啥叫中序呢，可能就是因为`callBack`的调用时机吧，下面吧上述遍历过程描述一下：

1.  `binarySearchTree.insert(19)`，这里`node`不是`null`，所以执行`this.inOrderTraverseNode(node.left,callBack)`;因为这里的`node.left`是8，所以再次进入该函数，`node`还不是`null`，再次进入`this.inOrderTraverseNode(node.left,callBack)`，因为8的节点左侧是`null`，所以不再递归执行，转而执行`callBack(node.key)`，因为`node.key`是8，所以首先打印的是8，
2.  打印完之后执行`this.inOrderTraverseNode(node.right,callBack)`，因为8的右侧是11，所以当再次执行`inOrderTraverseNode`函数是`node.left`是`null`，所以还是会执行`callBack(node.key)`，所以第二个是11
3.  然后执行`this.inOrderTraverseNode(node.right,callBack)`，此时11的右侧是`null`，所以直接跳了出来，也就是说最开始的那个`this.inOrderTraverseNode(node.left,callBack)`执行完了，之后执行`callBack(node.key)`，所以是19
4.  然后执行`this.inOrderTraverseNode(node.right,callBack)`，因为这里的`right`是22，所以再次执行`inOrderTraverseNode(node,callBack)`的时候，22的`left`是`null`，所以执行下面的`callBack(node.key)`，所以打印22。
5.  再次来到`this.inOrderTraverseNode(node.right,callBack)`，因为`node`是22，`node.right`是32，所以当以此为参数进入`inOrderTraverseNode(node,callBack)`的时候，因为`node.left`是`null`，所以来到`callBack(node.key)`，所以就会打印32，
6.  当走到`this.inOrderTraverseNode(node.right,callBack)`的时候由于`node.right`还是`null`，所以不执行啦。

同理

```js
// 前序排列
preOrderTraverse(callback) {
  this.preOrderTraverseNode(this.root, callback);
}
preOrderTraverseNode(node, callback) {
  if (node != null) {
    callback(node.key); // {1}
    this.preOrderTraverseNode(node.left, callback); // {2}
    this.preOrderTraverseNode(node.right, callback); // {3}
  }
}
```

 

```js

postOrderTraverse(callback) {
  this.postOrderTraverseNode(this.root, callback);
}
postOrderTraverseNode(node, callback) {
  if (node != null) {
    this.postOrderTraverseNode(node.left, callback); // {1}
    this.postOrderTraverseNode(node.right, callback); // {2}
    callback(node.key); // {3}
```

## 1.4、一个完整的树

```js
      // 创建一个树节点类
      class Node {
        constructor(key) {
          this.key = key;
          this.left = null;
          this.right = null;
        }
      }
      // 比较节点的值是否相同
      function compare(key, base) {
        if (key > base) {
          return "MORE_THAN";
        } else {
          return "LESS_THAN";
        }
      }

      class BinarySearchTree {
        constructor() {
          this.compareFn = compare;
          this.root = null; // 初始化根节点是null
        }
        // 插入节点
        insert(key) {
          if (this.root == null) {
            // 如果没有根节点那么首先插入到根节点的位置
            this.root = new Node(key);
          } else {
            // 如果有根节点则执行插入操作
            this.insertNode(this.root, key);
          }
        }
        // 这个实现要根据二叉树的特点
        insertNode(node, key) {
          if (this.compareFn(key, node.key) === "LESS_THAN") {
            // 如果要插入的节点比当前节点小
            if (node.left === null) {
              // 如果左侧是null，则直接插入到这个位置
              node.left = new Node(key);
            } else {
              this.insertNode(node.left, key); // 如果左侧有节点，则将node.left传入来继续寻找
            }
          } else {
            if (node.right === null) {
              // 如果右侧是null，就插入到右侧节点位置
              node.right = new Node(key);
            } else {
              this.insertNode(node.right, key); // 如果右侧有节点，则将node.right传入来继续寻找
            }
          }
        }
        // 1. 中序遍历
        inOrderTraverse(callBack) {
          this.inOrderTraverseNode(this.root, callBack);
        }
        // 真正的遍历操作
        inOrderTraverseNode(node, callBack) {
          // 如果当前节点存在
          if (node !== null) {
            this.inOrderTraverseNode(node.left, callBack);
            callBack(node.key);
            this.inOrderTraverseNode(node.right, callBack);
          }
        }
        // 前序排列
        preOrderTraverse(callback) {
          this.preOrderTraverseNode(this.root, callback);
        }
        preOrderTraverseNode(node, callback) {
          if (node != null) {
            callback(node.key); // {1}
            this.preOrderTraverseNode(node.left, callback); // {2}
            this.preOrderTraverseNode(node.right, callback); // {3}
          }
        }
        // 后序排列
        postOrderTraverse(callback) {
          this.postOrderTraverseNode(this.root, callback);
        }
        postOrderTraverseNode(node, callback) {
          if (node != null) {
            this.postOrderTraverseNode(node.left, callback); // {1}
            this.postOrderTraverseNode(node.right, callback); // {2}
            callback(node.key); // {3}
          }
        }
        // 寻找最小的节点,因为根据二叉树的特点，最左侧的树的节点永远是最小的
        min() {
          return this.minNode(this.root);
        }
        minNode(node) {
          let current = node;
          while (current !== null && current.left !== null) {
            current = current.left;
          }
          return current;
        }
        // 寻找最大节点
        max() {
          return this.maxNode(this.root);
        }
        // 同理，最右侧的树的节点是最大的
        maxNode(node) {
          let current = node;
          while (current !== null && current.right !== null) {
            current = current.right;
          }
          return current;
        }
        // 搜索特定值
        search(key) {
          return this.searchKeyNode(key, this.root);
        }
        searchKeyNode(key, node) {
          if (node !== null) {
            // 如果刚好相等直接返回节点
            if (node.key == key) {
              return node;
            } else if (node.key > key) {
              // 当前节点key>目标key，从左侧寻找
              return this.searchKeyNode(key, node.left);
            } else {
              // 当前节点key>目标key，从右侧寻找
              return this.searchKeyNode(key, node.right);
            }
          } else {
            return null;
          }
        }
        // 移除一个节点
        remove(key) {
          this.root = this.removeNode(this.root, key);
        }
        // 移除一个节点，其子节点也会被移除
        removeNode(node, key) {
          if (node == null) {
            // {2}
            return null;
          }
          if (this.compareFn(key, node.key) === "LESS_THAN") {
            // {3}
            node.left = this.removeNode(node.left, key); // {4}
            return node; // {5}
          } else if (this.compareFn(key, node.key) === "MORE_THAN") {
            // {6}
            node.right = this.removeNode(node.right, key); // {7}
            return node; // {8}
          } else {
            // 键等于 node.key
            // 第一种情况
            if (node.left == null && node.right == null) {
              // {9}
              node = null; // {10}
              return node; // {11}
            }
            // 第二种情况
            if (node.left == null) {
              // {12}
              node = node.right; // {13}
              return node; // {14}
            } else if (node.right == null) {
              // {15}
              node = node.left; // {16}
              return node; // {17}
            }
            // 第三种情况
            const aux = this.minNode(node.right); // {18}
            node.key = aux.key; // {19}
            node.right = this.removeNode(node.right, aux.key); // {20}
            return node; // {21}
          }
        }
      }
      let binarySearchTree = new BinarySearchTree();
      binarySearchTree.insert(19);
      binarySearchTree.insert(8);
      binarySearchTree.insert(22);
      binarySearchTree.insert(32);
      binarySearchTree.insert(11);
      console.log(binarySearchTree);

      function print(val) {
        console.log(val);
      }
      binarySearchTree.inOrderTraverse(print);

      console.log(binarySearchTree.min());
      console.log(binarySearchTree.max());
      console.log(binarySearchTree.search(12)); // null
      console.log(binarySearchTree.search(22)); // null
      binarySearchTree.remove(22);
      console.log(binarySearchTree);
```

## 二、AVL树

BST 存在一个问题：树的一条边可能会非常深；也就是说，树的一 条分支会有很多层，而其他的分支却只有几层 ，如下图所示：

![image.png](/images/92691af95040ec73.png)

 所以就有了**AVL树**，也叫自平衡树，其一大**特点就是**：**任何一个节点左右两侧子树的高度之差最多为 1**。

> 节点的高度是从节点到其任意子节点的边的最大值

在这个树中，需要对每个节点计算右子树高度（hr）和左子树高度（hl）之间的差值，该 值（hr－hl）应为 0、1 或-1。如果结果不是这三个值之一，则需要平衡该 AVL 树。

上面都提到一个词，叫**高度，**那么什么叫树的高度呢，把上面的话转义一下就是 ：

```plain
节点高度 = 子节点的最大高度+1
子节点的最大高度 = Math.max(左侧子节点高度，右侧子节点高度)+1
```

用代码实现就是：

```js
// AVL树
class AVLTree extends BinarySearchTree {
  constructor() {
    super();
  }
  // 获取某一节点的高度 = （子节点最大高度）+1，为啥加1，因为子节点到父节点还差一级呢
  getNodeHeight(node) {
    if (node === null) {
      return -1;
    } else {
      //根据上面的公式，要想计算高度，你就要计算子节点的最大高度，所以这里的关键就是怎么求子节点的最大高度，而每一个几点都有左右两个字子节点，所以，就有了
      /**
              * Math.max(
                this.getNodeHeight(node.left),
                this.getNodeHeight(node.right)
              ) + 1
               */
      // 算出的是当前子节点的最大高度，而子节点还有子节点，以此类推当然需要递归啦
      return (
        Math.max(
          this.getNodeHeight(node.left),
          this.getNodeHeight(node.right)
        ) + 1
      );
    }
  }
}
```
