## 一、定义

字典其实和之前的集合有相似之处，都是一个对象`{}`，只不过集合的结构是这样的`{a,b,c,d}`，其`key`和`value`都是一样的，而字典则是这种形式：`{key1:{key1,value1},key2:{key2,value2}}`。

## 代码示例

```js
// 将传入的item转化为字符串形式
      function toStrFn(item) {
        if (item === null) {
          return "null";
        } else if (item === undefined) {
          return "undefined";
        } else if (item instanceof String) {
          return item;
        } else {
          return item.toString();
        }
      }

      // 字典
      class Dictionary {
        constructor() {
          this.toStrFn = toStrFn; //  确保key是字符串形式
          this.table = {}; // 初始化字典
        }
        // 向字典中添加新的数据
        set(key, value) {
          if (key !== null && key !== undefined) {
            let keyStr = this.toStrFn(key);
            this.table[keyStr] = {
              key: keyStr,
              value: value,
            };
          }
        }
        // 判断是否已经有了某个key
        hasKey(key) {
          let keyStr = this.toStrFn(key);
          // 就看key对应的value是不是有值,有值就是对象, !!对象就是true
          return !!this.table[keyStr];
        }
        // 从字典中删除一个key
        remove(key) {
          if (this.hasKey(key)) {
            let keyStr = toStrFn(key);
            delete this.table[keyStr];
            return true;
          }
          return false;
        }
        // 从字典中查找一个值
        get(key) {
          if (this.hasKey(key)) {
            return this.table[this.toStrFn(key)];
          }
          return null;
        }
        // 返回由value组成的数组
        keyValues() {
          return Object.values(this.table);
        }
        // 返回由“健”组成的数组，真正的健
        keys() {
          return this.keyValues().map((item) => item.key);
        }
        // 返回由值组成的数组,真正的值
        values() {
          return this.keyValues().map((item) => item.value);
        }
        // 清空字典
        clear(){
          this.table = {}
        }
        // 返回字典大小
        size(){
          return this.keyValues().length
        }
        // 当前字典是否为空
        isEmpty(){
          return this.keyValues().length===0
        }
        // 转化为字符串
        toString(){
          if(this.isEmpty()){
            return ''
          }else{
            return this.keyValues().join(',')
          }
        }
      }
      let dictionary = new Dictionary();
      dictionary.set("hello", "哈喽");
      dictionary.set("world", "世界");
      dictionary.set("hello", "hola");
      console.log(dictionary.get("world"));
      console.log(dictionary);
      console.log(dictionary.keyValues());
      console.log(dictionary.keys());
      console.log(dictionary.toString());
```
