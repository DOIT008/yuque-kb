FILO（**先入后出**）,常见操作就是出栈（就是将栈顶元素扔出去）和入栈（往栈顶添加元素）

### 基于数组

## 代码示例

```js
class Stack {
  constructor(){
    this.stack = []
  }
  //移除栈顶元素，同时返回被移除的元素
  pop(){
    return this.stack.shift()
  }
  // 返回栈顶的元素，仅仅返回
  peek(){
    return this.stack[0]
  }
  // 添加元素
  push(node){
    this.stack.unshift(node)
  }
  // 是否为空
  isEmpty(){
    return !this.stack.length
  }
  // 清空栈中的所有元素
  clear(){
    this.stack = []
  }
  // 返回栈的大小
  size(){
    return this.stack.length
  }
}
let stack = new Stack();
stack.push(12);
stack.push(19);
stack.push(90);
stack.push(79);
console.log(stack);
console.log(stack.size());
console.log(stack.pop());
console.log(stack.size());
console.log(stack.isEmpty());
```

### 基于对象

```js
class StackIndex {
  constructor(){
    this.count = 0;
    this.stack = {}
  }
  // 添加元素
  push(node){
    this.stack[this.count] = node;
    this.count++
  }
  // 移除栈顶元素，同时返回被移除的元素
  pop(){
    if(this.isEmpty()){
      return undefined
    }
    this.count--
    const result = this.stack[this.count];// 保存要被删除的值
    delete this.stack[this.count];//删除对应的值
    return result //返回值
  }
  // 是否为空
  isEmpty(){
    return this.count===0
  }
  // 清空所有元素
  clear(){
    this.stack = {}
    this.count = 0;
  }
  // 栈大小
  size(){
    return this.count
  }
  // 返回栈顶元素，其实就是返回最后一个push的元素
  peek(){
    if(this.count===0){
      return undefined
    }
    return this.stack[this.count-1]// 因为是从0 开始的
  }
  // toString方法
  toString(){
    if(this.isEmpty()){
      return ''
    }else{
      let str = '';
      for (const key in this.stack) {
        str+=`${this.stack[key]},`
        console.log(key);
      };
      return str.substr(0,str.length-1)// 去掉最后一额‘,’
    }
  }
}
let stackIndex = new StackIndex();
stackIndex.push("H");
stackIndex.push("E");
stackIndex.push("L");
stackIndex.push("L");
stackIndex.push("O");
stackIndex.push("W");
stackIndex.push("O");
stackIndex.push("R");
stackIndex.push("L");
stackIndex.push("D");
stackIndex.pop();
stackIndex.pop();
stackIndex.pop();
stackIndex.pop();
console.log(stackIndex.peek());
console.log(stackIndex.size());
console.log(stackIndex.isEmpty());
console.log(stackIndex);
console.log(stackIndex.toString());
```

## 刷题

对前端而言，栈在某种程度上来说就是用**数组来进行模拟（创建栈就是创建数组）**，但是栈只有两种上述的几个方法，所以在刷题的时候也要用数组的方法比如`pop`，`push`等来模拟上述栈的行为，不能用诸如`find`，`findIndex`，`some`等比较高级的方法。
